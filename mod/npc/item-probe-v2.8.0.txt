// Item Repair Hub v2.8.0: read-only ID/name baseline check and era query; no app integration.
-	script	IRHNativeProbeV21	-1,{
OnInit:
    bindatcmd "irhprobe", strnpcinfo(3) + "::OnProbe", 0, 99;
    end;
OnProbe:
    .@req$ = .@atcmd_parameters$[0];
    // "era": which era this server runs, so the editor loads the matching items and skills.
    if (.@atcmd_parameters$[1] == "era") {
        dispbottom "@@reply " + .@req$ + " 1/1 OK|era|" + (checkre(0) ? "renewal" : "pre-renewal");
        end;
    }
    .@id = atoi(.@atcmd_parameters$[1]);
    if (.@id <= 0 || getitemname(.@id) == "null") {
        dispbottom "@@reply " + .@req$ + " 1/1 MISSING";
        end;
    }
    dispbottom "@@reply " + .@req$ + " 1/1 OK|" + .@id + "|" + getiteminfo(.@id, ITEMINFO_AEGISNAME);
    end;
}
